6. 自连接（SELF JOIN）：自己连自己
 
概念
 
把同一张表，当成两张表来连接，用来处理表内的关联关系（比如邀请人、上下级这种自己关联自己的
场景）。
案例：查询每个玩家，以及他的邀请人是谁
 
查询结果
 
-- 左连接结果
SELECT wj.wanjia_nicheng, yx.yingxiong_ming
FROM wanjia wj
LEFT JOIN zhanji zj ON wj.wanjia_id = zj.wanjia_id
LEFT JOIN yingxiong yx ON zj.yingxiong_id = yx.yingxiong_id
UNION -- 合并去重
-- 右连接结果
SELECT wj.wanjia_nicheng, yx.yingxiong_ming
FROM wanjia wj
RIGHT JOIN zhanji zj ON wj.wanjia_id = zj.wanjia_id
RIGHT JOIN yingxiong yx ON zj.yingxiong_id = yx.yingxiong_id;
SELECT 
  u.wanjia_nicheng AS 玩家昵称,
  inviter.wanjia_nicheng AS 邀请人昵称
FROM wanjia u
-- 玩家表自己连自己：u是当前玩家，inviter是邀请人
LEFT JOIN wanjia inviter ON u.yaoqingren_id = inviter.wanjia_id;
