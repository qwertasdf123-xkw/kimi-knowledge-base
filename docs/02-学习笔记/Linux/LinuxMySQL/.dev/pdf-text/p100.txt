#多个LIKE
mysql> select id,name from stu where name like '%n%' or name like '%i%';

| id | name     |

| 11 | xiaohong |
| 12 | zhangsan |

| 18 | xiaohong |
| 19 | xiaohong |
| 20 | zhangsan |

#连续
mysql> select id,name from stu where id between 12 and 15;
+----+----------+
| id | name     |
+----+----------+
| 12 | zhangsan |
| 13 | xiaozhou |
| 14 | test1    |
| 15 | test2    |
